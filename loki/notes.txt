Instructions:
- Deploy Mimir:
    kubectl create namespace mimir
    helm -n mimir install mimir grafana/mimir-distributed
- Deploy Loki:
    kubectl create namespace loki
    helm -n loki install loki grafana/loki --values loki-values.yaml
- Deploy Grafana:
    kubectl create namespace grafana
    helm -n grafana install grafana grafana/grafana
- Deploy Alloy:
    kubectl create namespace alloy
    helm -n alloy install alloy grafana/alloy --values alloy-values.yaml
- Port-forward Grafana port:
    export POD_NAME=$(kubectl get pods --namespace grafana -l "app.kubernetes.io/name=grafana,app.kubernetes.io/instance=grafana" -o jsonpath="{.items[0].metadata.name}")
    kubectl --namespace grafana port-forward $POD_NAME 3000 &
- Get Grafana admin password:
    kubectl get secret --namespace grafana grafana -o jsonpath="{.data.admin-password}" | base64 --decode ; echo
- Install Explore Logs plugin:
    Browse to http://localhost:3000 and login as admin
    Browse to http://localhost:3000/plugins/grafana-lokiexplore-app and click "Install"
- Create Loki datasource:
    Browse to http://localhost:3000/connections/add-new-connection and click "Loki"
    Click "Add new data source"
    Enter http://loki-gateway.loki.svc.cluster.local for URL
    Add HTTP header "X-Scope-OrgId" with value "foo"
    Click "Save & test"
- Explore logs:
    Browse to http://localhost:3000/a/grafana-lokiexplore-app/explore

Ingest logs from Mac host:
- Install Alloy
    brew tap grafana/grafana
    brew install grafana/grafana/alloy
- Run Alloy
    brew services start alloy
- Configure Alloy
    vi $(brew --prefix)/etc/alloy/config.alloy
    --------
    logging {
      level  = "info"
      format = "logfmt"
    }
    
    local.file_match "local_files" {
      path_targets = [{"__path__" = "/Users/anderl01/projects/osiks-public/loki/lab.log"}]
      sync_period = "5s"
    }
    
    loki.source.file "log_scrape" {
      targets       = local.file_match.local_files.targets
      forward_to    = [loki.write.grafana_loki.receiver]
      tail_from_end = false
    }
    
    loki.write "grafana_loki" {
      endpoint {
        url = "http://localhost:3100/loki/api/v1/push"
        tenant_id = "mac"
    
        // basic_auth {
        //  username = "admin"
        //  password = "admin"
        // }
      }
    }
    -------
    brew services restart alloy
